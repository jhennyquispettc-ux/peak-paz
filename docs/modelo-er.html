  disponibilidad_semanal {

  }
  bloqueo_fecha {

    uuid perfil_guia_id FK
    date fecha_desde
    date fecha_hasta
    varchar motivo
  }
  salida_grupal {
    uuid id PK
    uuid agencia_id FK
    uuid variante_id FK
    date fecha
    time hora_encuentro
    varchar punto_encuentro
    numeric precio_por_persona
    smallint cupo_minimo
    smallint cupo_maximo
    varchar nombre_guia
    enum estado "ver E1"
    timestamptz cierre_en
  }
  reserva {
    uuid id PK
    varchar codigo UK
    enum tipo "PRIVADA | GRUPAL"
    enum origen "WEB | MANUAL"
    uuid turista_id FK
    uuid perfil_guia_id FK
    uuid salida_grupal_id FK
    uuid variante_id FK
    date fecha_tour
    time hora_inicio
    smallint cantidad_personas
    varchar contacto_manual
    numeric monto_total
    enum estado "ver E2"
    timestamptz responder_antes_de
    timestamptz pagar_antes_de
  }
  pago {
    uuid id PK
    uuid reserva_id FK
    varchar pasarela
    varchar id_externo_orden
    varchar id_externo_autorizacion
    varchar id_externo_captura
    numeric monto
    char moneda
    numeric comision_pasarela
    enum estado "ver E4"
  }
  reembolso {
    uuid id PK
    uuid pago_id FK
    numeric monto
    enum motivo "ver E3"
    varchar id_externo
    enum estado "PENDIENTE | COMPLETADO | FALLIDO"
  }
  liquidacion {
    uuid id PK
    uuid reserva_id FK,UK
    uuid beneficiario_id FK
    numeric monto_bruto
    numeric porcentaje_comision
    numeric monto_comision
    numeric descuento_penalizacion
    numeric monto_neto
    enum estado "PENDIENTE | PAGADA"
  }
  reprogramacion {
    uuid id PK
    uuid reserva_id FK
    date fecha_anterior
    date fecha_nueva
    varchar motivo
    enum respuesta "PENDIENTE | ACEPTA | RECHAZA"
    timestamptz respondido_en
  }
E1 salida_grupal.estadoPUBLICADA, CONFIRMADA, EN_CURSO, COMPLETADA, CANCELADA_MINIMO, CANCELADA_AGENCIA, CANCELADA_CLIMA
E2 reserva.estadoSOLICITADA, ACEPTADA, CONFIRMADA, EN_CURSO, COMPLETADA, RECHAZADA, EXPIRADA, CANCELADA_TURISTA, CANCELADA_PROVEEDOR, CANCELADA_CLIMA
E3 reembolso.motivoCANCELACION_TURISTA, CANCELACION_PROVEEDOR, CLIMA, REPROGRAMACION_RECHAZADA, RECLAMO
E4 pago.estadoPENDIENTE, RETENIDO, COBRADO, ANULADO, REEMBOLSADO, FALLIDO
Regla reserva.tipoPRIVADA exige perfil_guia_id y deja salida_grupal_id vacío. GRUPAL exige salida_grupal_id y deja perfil_guia_id vacío. Origen MANUAL solo existe en GRUPAL y no tiene turista_id.
